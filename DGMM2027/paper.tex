% Float placement. LaTeX's defaults are tuned for floats that are a small
% fraction of the page and are far too conservative for Sec. 6, where the
% figures are large but no longer full-page: with the stock values a 6in
% figure* is pushed onto a float-only page and the remaining ~4in is left
% blank rather than being filled with text or a second figure. Raising the
% fractions lets a big float share its page, and lowering \textfraction
% stops LaTeX from insisting on a minimum quantity of body text per page.
\renewcommand{\topfraction}{0.9}        % max page fraction for top floats
\renewcommand{\bottomfraction}{0.8}
\renewcommand{\textfraction}{0.07}      % min page fraction that must be text
\renewcommand{\floatpagefraction}{0.75} % a float page must be >=75% full
\renewcommand{\dbltopfraction}{0.9}         % same, for two-column figure*
\renewcommand{\dblfloatpagefraction}{0.75}

\setcounter{dbltopnumber}{3}
